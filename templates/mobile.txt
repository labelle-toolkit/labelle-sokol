export fn frame() callconv(.c) void {
    if (!g.isRunning()) {
        window.requestQuit();
        return;
    }
    frame_count +|= 1;

    const actual_w = window.width();
    const actual_h = window.height();
    BackendGfxType.setScreenSize(actual_w, actual_h);
    // Tell the gfx backend the design canvas size so screen-space sprites
    // (background, UI) map design coords (0..screen_w, 0..screen_h) to the
    // full physical screen NDC range. Must be called after setScreenSize.
    BackendGfxType.setDesignSize(@intCast(screen_w), @intCast(screen_h));
    g.setScreenHeight(@as(f32, @floatFromInt(screen_h)));
    // On the first frame, scale camera zoom so the design height fills the physical
    // screen height exactly. With high_dpi=true, actual_h is the framebuffer height.
    // zoom = actual_h / design_h maps one design pixel to (actual_h/design_h) physical
    // pixels, giving sharp high-DPI rendering without layout changes to game code.
    if (frame_count == 1) {
        const zoom = @as(f32, @floatFromInt(actual_h)) / @as(f32, @floatFromInt(screen_h));
        g.getCamera().setZoom(zoom);
    }
    // Cap dt at 4 frames to prevent physics explosions after backgrounding.
    // @max(target_fps, 1) guards against a target_fps=0 (unlimited/vsync-only) config dividing by zero.
    const dt: f32 = @min(@as(f32, @floatCast(window.frameDuration())), 4.0 / @as(f32, @floatFromInt(@max(target_fps, 1))));
{{preview_heartbeat}}{{tick_code}}    g.tick(dt);
{{preview_pre_render}}    const pass_action = window.beginFrame();
    window.beginPass(pass_action);
    g.render();
    g.renderGizmos();
    // Flush queued sokol-gl primitives BEFORE the GUI renders so imgui
    // draws are submitted on top of sprites in the same pass. Same shape

// {{entry_comment}}
// On Android sokol's C runtime provides the entry point
// (ANativeActivity_onCreate) and calls sokol_main() to configure the app, so
// sokol_main must RETURN sapp_desc and must NOT call sapp_run() itself.
// iOS is different: sokol-zig compiles sokol_app with SOKOL_NO_ENTRY on every
// non-Android target (sokol_defines.h), so sokol_app has no C main() there and
// instead defines its OWN stub `sokol_main` for the linker. The iOS exe
// therefore starts in Zig's `main` below, which runs this descriptor, and
// `sokol_main` is exported on Android only (exporting it on iOS is a duplicate
// symbol with the stub). labelle-cli#471 I0. Android builds a shared library,
// so `main` is never referenced there.
// Zig 0.16's default Mach-O `std.debug.SelfInfo` calls
// `_dyld_get_image_header_containing_address`, which the iOS SDK does not
// export, so any panic/stack-trace path fails to LINK on iOS. `void` opts out
// of self-debug-info there (traces print without symbols); other targets keep
// the default (`root.debug` has no `SelfInfo`).
pub const debug = if (@import("builtin").target.os.tag == .ios) struct {
    pub const SelfInfo = void;
} else struct {};

comptime {
